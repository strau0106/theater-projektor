
# ---------------------------------------------------------------- projektor
# Appended by the theater-projektor image build (image/stage-projektor).
# Display: full KMS (dtoverlay=vc4-kms-v3d above, with cma-256: scanout + decoder
# buffers). Do NOT raise CMA/gpu_mem on 1 GB boards: GL (v3d) buffers live in normal
# RAM, cma-512 + gpu_mem=256 left ~236 MB and the app was OOM-killed on a Pi 4 1 GB.
# The KMS mode itself is forced in cmdline.txt
# (video=HDMI-A-<port>:1920x1080@60D, HDMI_PORT in secrets/build.env) so a beamer switched on
# after boot gets a picture.
[all]
# firmware stage: treat both HDMI ports as connected even without a display/EDID
hdmi_force_hotplug:0=1
hdmi_force_hotplug:1=1
# no rainbow splash
disable_splash=1
# no audio needed (no sound card driver, slightly faster boot)
dtparam=audio=off
# Bluetooth off: not used; frees the UART and avoids 2.4 GHz coexistence with WiFi
# (BT and WiFi share one radio/antenna on Pi 4/5)

[pi4]
dtoverlay=disable-bt
# H.264 hardware decoder (bcm2835-codec) needs some firmware memory; 128 is plenty
gpu_mem=128

[pi5]
dtoverlay=disable-bt-pi5

[all]
# ---------------------------------------------------------------- /projektor
